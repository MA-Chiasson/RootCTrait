; ===================== ROOTCTRAIT MULTI-BATCH CONFIGURATION =====================
; Lines starting with ';' are comments. After a setting, ' ; text' is also a comment.

; Batches to process: ALL (every batch defined below) or a comma separated list,
; e.g. BATCHES=block1_t1,block2_t1
BATCHES=ALL

; Root folders: input masks are read from DATA_ROOT/<batch>/ and results are
; written to RESULTS_ROOT/<batch>/
DATA_ROOT=data
RESULTS_ROOT=results

; =========== Settings SHARED by all batches (keeps traits comparable) ===========
; Voxel size in mm, in the order depth, X, Z
VOXEL=0.39,0.39,0.2
; Minimum length (skeleton voxels) of a spur kept during pruning
PRUNE_VOX=5
; Segments shorter than this length (mm) are discarded
MIN_SEG_LEN_MM=2.0
; Sheet rule: a segment is removed when bc >= BC_MIN, lin < LIN_MAX and length < LEN_MAX (mm)
BC_MIN=3
LIN_MAX=0.7
LEN_MAX=15
; Dense rule: a segment is also removed when the skeleton around it is dense
; (DENS_MAX skeleton voxels or more within 2 mm) and it is shorter than DENS_LEN_MAX (mm).
; DENS_MAX=0 disables the dense rule.
DENS_MAX=35
DENS_LEN_MAX=6
; Detached fragments at least RESCUE_MIN_MM long (mm) and not dense are kept as roots
; and joined back to the system; 0 disables the rescue.
RESCUE_MIN_MM=5
; 1 = after decontamination, keep only the component that contains the primary root
DROP_ORPHANS=1
; 1 = write one interactive HTML figure per sample (requires plotly)
SAVE_FIGURES=0
; Time limit per sample, in seconds
TIMEOUT=1800

; Number of samples processed in parallel. 1 = serial processing (default).
; Validate on a single batch before increasing it (the per sample time limit is kept).
PARALLEL=1

; ========= Batch definitions: BATCH name | file pattern | axis_order (optional) =========
; name        : must match EXACTLY the subfolder in DATA_ROOT
; file pattern: file name with {name} standing for the sample identifier,
;               e.g. block1_t1_{name}.mat matches block1_t1_S001.mat, block1_t1_S002.mat, ...
; axis_order  : optional axis permutation (as numpy.transpose) that puts depth first,
;               e.g. 2,1,0 for a TIFF stack whose last axis is depth; leave empty for .mat files
; (the French keyword BLOC is accepted as an alias of BATCH)
BATCH block1_t1 | block1_t1_{name}.mat |
BATCH block2_t1 | block2_t1_{name}.mat |
BATCH block3_t1 | block3_t1_{name}.mat |
BATCH block1_t2 | block1_t2_{name}.mat |
BATCH block2_t2 | block2_t2_{name}.mat |
BATCH block3_t2 | block3_t2_{name}.tif | 2,1,0
